class Solution {
public:
    vector<string> findpath(TreeNode* root){
        if (root == nullptr) return {};
        if (root->left == nullptr && root->right == nullptr){
            return {to_string(root->val)};
        }
        vector<string> path, leftPath, rightPath;
        leftPath = findpath(root->left);
        rightPath = findpath(root->right);
        path.insert(path.end(), leftPath.begin(), leftPath.end());
        path.insert(path.end(), rightPath.begin(), rightPath.end());
        for (auto &p:path){
            p = to_string(root->val) + "->" + p;
        }
        return path;      

    }
    vector<string> binaryTreePaths(TreeNode* root) {
        vector<string> ans;
        ans = findpath(root);
        return ans;
    }
};
