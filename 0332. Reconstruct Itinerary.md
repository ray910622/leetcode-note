## Initial Solution
- Use `int` in `flightMap` to record the `[from, to]` matrix with `string` index.
```cpp
class Solution {
public:
    vector<string> ans;
    unordered_map<string, map<string,int>> flightMap;

    void dfs(const string& from) {
        for(pair<const string,int>& flight: flightMap[from]){
            if (flight.second > 0 ){
                flight.second--;
                dfs(flight.first);
                ans.push_back(flight.first);
            }
        }
    }

    vector<string> findItinerary(vector<vector<string>>& tickets) {
        for (vector<string> &ticket: tickets){
            flightMap[ticket[0]][ticket[1]]++;
        }
        dfs("JFK");
        ans.push_back("JFK");
        reverse(ans.begin(),ans.end());
        return ans;
    }
};
```
