## Initial Solution
- From top to bottom.
```cpp
class Solution {
public:
    TreeNode* ans;
    bool isDescendant(TreeNode* root, TreeNode* target) {
        if (root == nullptr)
            return false;
        if (root == target)
            return true;
        return isDescendant(root->left, target) ||
               isDescendant(root->right, target);
    }
    void traversal(TreeNode* root, TreeNode* p, TreeNode* q) {
        if (root == nullptr) return;
        if (isDescendant(root, p) && isDescendant(root, q)) {
            ans = root;
        }
        traversal(root->left, p, q);
        traversal(root->right, p, q);
    }
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        traversal(root, p, q);
        return ans;
    }
};
```
- Time Complexity: $O(n^2)$ .

## Second Solution
- From bottom to top, use backpropagation.
```cpp
class Solution {
public:
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        if (root == nullptr)
            return nullptr;
        if (root == p || root == q) return root;

        TreeNode *leftResult = lowestCommonAncestor(root->left, p, q),
            *rightResult = lowestCommonAncestor(root->right, p, q);
        
        if (leftResult == nullptr) return rightResult;
        if (rightResult == nullptr) return leftResult;
        return root;
    }
};
```
- Time Complexity: $O(n^2)$ .
