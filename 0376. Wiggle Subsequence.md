class Solution {
public:
    int wiggleMaxLength(vector<int>& nums) {
        if (nums.size() <= 1)
            return nums.size();
        int ans = 1;
        vector<int> diff_nums;
        for (int i = 1; i < nums.size(); i++) {
            diff_nums.push_back(nums[i] - nums[i - 1]);
        }
        int status = 0;
        for (int i = 0; i < diff_nums.size(); i++) {
            if (((status == 0 && diff_nums[i] != 0)) ||
                diff_nums[i] * status < 0) {
                ans++;
                status = diff_nums[i];
            }
        }
        return ans;
    }
};
