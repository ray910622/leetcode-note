class Solution {
public:
    vector<int> cur_set;
    vector<vector<int>> ans;
    void backtracking(int k, int n, int begin_num){
        if (n<0 || k> 10 - begin_num) return;
        if (k == 0){
            if (n == 0) ans.push_back(cur_set);
            return;
        }
        for (int i = begin_num ; i<=9 ; i++){
            cur_set.push_back(i);
            backtracking( k -1, n -i , i+1);
            cur_set.pop_back();
        }

    }
    vector<vector<int>> combinationSum3(int k, int n) {
        backtracking(k, n ,1);
        return ans;
    }
};
