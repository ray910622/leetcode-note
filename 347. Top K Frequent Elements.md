## Initial Solution
```cpp
class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        vector<int> ans;
        unordered_map<int, int> freq;
        for (int i = 0; i < nums.size(); i++) {
            freq[nums[i]]++;
        }
        stack<int> k_stk, reserve_stk;

        for (auto& [num, count] : freq) {
            while (!k_stk.empty() && freq[num] > freq[k_stk.top()]) {
                reserve_stk.push(k_stk.top());
                k_stk.pop();
            }
            if (k_stk.size() < k) {
                k_stk.push(num);
            }
            while (!reserve_stk.empty() && k_stk.size() < k) {
                k_stk.push(reserve_stk.top());
                reserve_stk.pop();
            }
        }

        for (int i = 0; i < k; i++) {
            ans.push_back(k_stk.top());
            k_stk.pop();
        }
        return ans;
    }
};
```

## Second Solution

```cpp
class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        class Cmp{
            public: 
                bool operator()(pair<int,int> a, pair<int,int> b){
                    return a.second > b.second;
                }
        };
        vector<int> ans;
        unordered_map<int, int> freq;
        for (int i = 0; i < nums.size(); i++) {
            freq[nums[i]]++;
        }
        priority_queue<pair<int,int>, vector<pair<int,int>>, Cmp> min_heap;

        for (auto& [num, count] : freq) {
            if (min_heap.size() < k){
                min_heap.push({num,count});
            }
            else if (min_heap.top().second < count){
                min_heap.pop();
                min_heap.push({num,count});
            }
        }

        for (int i = 0; i < k; i++) {
            ans.push_back(min_heap.top().first);
            min_heap.pop();
        }
        return ans;
    }
};
```
