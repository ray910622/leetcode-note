## Initial Solution
```cpp
class Solution {
public:
    int maxDepth(TreeNode* root) {
        if (root == nullptr) {
            return 0;
        }
        return 1 + max(maxDepth(root->left), maxDepth(root->right));
    }
    int findValue(TreeNode* root, int depth){
        int val;
        if (maxDepth(root) == 1 && root->left ==nullptr && root->right == nullptr) return root->val;
        if (maxDepth(root->left) == depth-1){
            val = findValue(root->left, depth-1);
        }else{
            val = findValue(root->right, depth-1);
        }
        return val;
    }
    int findBottomLeftValue(TreeNode* root) {
        return findValue(root, maxDepth(root));
    }
};
```
- Time Complexity: $O(n^2)$.

## Second Solution
```cpp
class Solution {
public:
    int maxDepth = INT_MIN, ans;

    void traversal(TreeNode* root, int depth) {
        if (root->left == nullptr && root->right == nullptr) {
            if (depth > maxDepth) {
                maxDepth = depth;
                ans = root->val;
            }
            return;
        }
        if (root->left != nullptr) {
            depth++;
            traversal(root->left, depth);
            depth--;
        }
        if (root->right != nullptr) {
            depth++;
            traversal(root->right, depth);
            depth--;
        }
    }
    int findBottomLeftValue(TreeNode* root) {
        traversal(root, 0);
        return ans;
    }
};
```

- Time Complexity: $O(n)$.
