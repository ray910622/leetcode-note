class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        vector<int> ans;
        deque<int> dq;
        int l = 1 - k, r = 0;

        for (; r < nums.size(); l++, r++) {
            while (!dq.empty() && dq.front() < l)
                dq.pop_front();
            while (!dq.empty() && nums[r] > nums[dq.back()]) {
                dq.pop_back();
            }
            dq.push_back(r);
            if (r >= k - 1)
                ans.push_back(nums[dq.front()]);
        }
        return ans;
    }
};
