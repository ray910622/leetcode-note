class Solution {
public:
    vector<vector<int>> combine(int n, int k) {
        if (n < k)
            return {};
        if (k == 0) {
            return {{}};
        }
        vector<vector<int>> choice = combine(n - 1, k - 1),
                            not_choice = combine(n - 1, k);
        for (auto& nums : choice) {
            nums.push_back(n);
        }
        choice.insert(choice.end(), not_choice.begin(), not_choice.end());
        return choice;
    }
};
