class Solution {
public:
    int findInorderIndex(vector<int>& vec, int low, int high, int target) {
        for (int i = low; i <= high; i++) {
            if (vec[i] == target) {
                return i;
            }
        }
        return -1;
    }
    TreeNode* buildTreeIdx(vector<int>& inorder, vector<int>& preorder,
                           int inLeft, int inRight, int preLeft,
                           int preRight) {
        // closed interval
        if (preLeft > preRight) {
            return nullptr;
        }

        int rootVal = preorder[preLeft];
        int rootIdx = findInorderIndex(inorder, inLeft, inRight, rootVal);
        int leftTreeSize = rootIdx - inLeft, rightTreeSize = inRight - rootIdx;
        TreeNode* root = new TreeNode(rootVal);

        root->left = buildTreeIdx(inorder, preorder, inLeft, rootIdx - 1,
                                  preLeft +1 , preLeft + leftTreeSize);
        root->right = buildTreeIdx(inorder, preorder, rootIdx + 1, inRight,
                                   preRight - rightTreeSize +1 , preRight);

        return root;
    }

    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {

        return buildTreeIdx(inorder, preorder, 0, inorder.size() - 1, 0,
                            preorder.size() - 1);
    }
};
