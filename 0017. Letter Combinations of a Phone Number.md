## Iniitial Solution 
```cpp
class Solution {
public:
    vector<char> mapping(char num) {

        if (num == '7')
            return {'p', 'q', 'r', 's'};
        if (num == '8')
            return {'t', 'u', 'v'};
        if (num == '9')
            return {'w', 'x', 'y', 'z'};

        // case '1' ~ '6'
        vector<char> res;
        char c = (num - '2') * 3 + 'a';
        for (int i = 0; i < 3; i++) {
            res.push_back(c);
            c++;
        }
        return res;
    }
    string path;
    vector<string> ans;
    void backtracking(string digits, int beginIdx) {
        if (beginIdx == digits.size()) {
            ans.push_back(path);
            return;
        }
        vector<char> char_list = mapping(digits[beginIdx]);
        for (auto c : char_list) {
            path.push_back(c);
            backtracking(digits, beginIdx + 1);
            path.pop_back();
        }
    }
    vector<string> letterCombinations(string digits) {
        backtracking(digits, 0);
        return ans;
    }
};
```

## Second Solution
- Reduce space.
```cpp
class Solution {
private:
    const string mapping[10] =
        {"" ,"", "abc", "def","ghi","jkl","mno","pqrs","tuv","wxyz" };

public:
    string path;
    vector<string> ans;
    void backtracking(string& digits, int beginIdx) {
        if (beginIdx == digits.size()) {
            ans.push_back(path);
            return;
        }
        int digit = digits[beginIdx] - '0' ;
        for (auto c :  mapping[digit]) {
            path.push_back(c);
            backtracking(digits, beginIdx + 1);
            path.pop_back();
        }
    }
    vector<string> letterCombinations(string digits) {
        backtracking(digits, 0);
        return ans;
    }
};
```
