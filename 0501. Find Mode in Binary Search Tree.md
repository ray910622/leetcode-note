## Initial Solution
```cpp
class Solution {
public:
    struct Comp {
        bool operator()(pair<int, int> a, pair<int, int> b) {
            return a.second < b.second;
        }
    };
    void traversal(TreeNode* root, unordered_map<int, int>& nums) {
        if (root == nullptr)
            return;
        traversal(root->left, nums);
        nums[root->val]++;
        traversal(root->right, nums);
    }
    vector<int> findMode(TreeNode* root) {
        unordered_map<int, int> nums;
        vector<int> ans;
        traversal(root, nums);
        priority_queue<pair<int, int>, vector<pair<int, int>>, Comp> heap(
            nums.begin(), nums.end());

        int times = heap.top().second;
        while (!heap.empty() && heap.top().second == times) {
            ans.push_back(heap.top().first);
            heap.pop();
        }
        return ans;
    }
};
```

## Second Solution
```cpp
class Solution {
public:
    TreeNode* prev = nullptr;
    int count = 0, max_count = 1;
    vector<int> ans;

    void traversal(TreeNode* cur) {
        if (cur == nullptr)
            return;
        traversal(cur->left);
        if (prev != nullptr && cur->val == prev->val) {
            count++;
        } else {
            count = 1;
        }

        if (count == max_count) {
            ans.push_back(cur->val);
        }
        if (count > max_count) {
            ans = {cur->val};
            max_count = count;
        }

        prev = cur;
        traversal(cur->right);
    }
    vector<int> findMode(TreeNode* root) {
        traversal(root);
        return ans;
    }
};
```
