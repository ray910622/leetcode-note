## Initial Solution
- The `attacked` function use time complexity `O(n^2)`.
```cpp
class Solution {
public:
    vector<vector<string>> ans;
    vector<string> board;
    bool attacked(int row, int col) {
        for (int i = 0; i < board.size(); i++) {
            for (int j = 0; j < board.size(); j++) {
                if (board[i][j] == 'Q') {
                    if (i + j == row + col)
                        return true;
                    if (i - j == row - col)
                        return true;
                    if (i == row)
                        return true;
                    if (j == col)
                        return true;
                }
            }
        }
        return false;
    }
    void backtracking(int n, int row) {
        if (n == 0) {
            ans.push_back(board);
            return;
        }
        for (int i = 0; i < board.size(); i++) {
            if (attacked(row, i))
                continue;
            board[row][i] = 'Q';
            backtracking(n - 1, row + 1);
            board[row][i] = '.';
        }
    }
    void initialize(int n) {
        for (int i = 0; i < n; i++) {
            string row = "";
            for (int j = 0; j < n; j++) {
                row.push_back('.');
            }
            board.push_back(row);
        }
    }
    vector<vector<string>> solveNQueens(int n) {
        initialize(n);
        backtracking(n, 0);
        return ans;
    }
};
```

## Second Solution
- Use three `used` vector to reduce `attacked` time complexity to `O(1)`.
```cpp
class Solution {
public:
    vector<vector<string>> ans;
    vector<string> board;
    vector<bool> colUsed;
    vector<bool> diagonalUsed;
    vector<bool> antiDiagonalUsed;
    bool attacked(int row, int col) {
        return (colUsed[col] || diagonalUsed[row - col + board.size()] ||
                antiDiagonalUsed[row + col]);
    }
    void place(int n, int row, int col, bool v) {
        board[row][col] = v ? 'Q' : '.';
        colUsed[col] = v;
        diagonalUsed[row - col + n] = v;
        antiDiagonalUsed[row + col] = v;
    }
    void backtracking(int n, int row) {
        if (row == n) {
            ans.push_back(board);
            return;
        }
        for (int i = 0; i < board.size(); i++) {
            if (attacked(row, i))
                continue;
            place(n, row, i, true);
            backtracking(n, row + 1);
            place(n, row, i, false);
        }
    }
    void initialize(int n) {
        board.assign(n, string(n, '.'));
        colUsed.assign(n, false);
        diagonalUsed.assign(n + n, false);
        antiDiagonalUsed.assign(n + n, false);
    }
    vector<vector<string>> solveNQueens(int n) {
        initialize(n);
        backtracking(n, 0);
        return ans;
    }
};
```
