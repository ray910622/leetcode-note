class Solution {
public:
    bool isInRange(TreeNode* root, long low, long high) {
        if (root == nullptr)
            return true;
        if (root->val > low && root->val < high)
            return (isInRange(root->left, low, root->val) &&
                    isInRange(root->right, root->val, high));
        else
            return false;
    }
    bool isValidBST(TreeNode* root) {
        return isInRange(root, LONG_MIN, LONG_MAX);
    }
};
