class Solution {
public:
    vector<vector<bool>> row_used =
        vector<vector<bool>>(9, vector<bool>(10, false));

    vector<vector<bool>> col_used =
        vector<vector<bool>>(9, vector<bool>(10, false));

    vector<vector<bool>> grid_used =
        vector<vector<bool>>(9, vector<bool>(10, false));

    bool isValid(const vector<vector<char>>& board, int row, int col, int num) {
        int grid_index = (row / 3) * 3 + col / 3;
        if (row_used[row][num] || col_used[col][num] ||
            grid_used[grid_index][num])
            return false;

        return true;
    }
    void place(vector<vector<char>>& board, int row, int col, int num,
               bool status) {
        int grid_index = (row / 3) * 3 + col / 3;

        board[row][col] = status ? '0' + num : '.';
        row_used[row][num] = status;
        col_used[col][num] = status;
        grid_used[grid_index][num] = status;
    }
    bool dfs(vector<vector<char>>& board, int box) {
        if (box == 81) {
            return true;
        }
        int row = box / 9, col = box % 9;
        if (board[row][col] == '.') {
            for (int num = 1; num <= 9; num++) {
                if (isValid(board, row, col, num)) {
                    place(board, row, col, num, true);
                    if (dfs(board, box + 1))
                        return true;
                    place(board, row, col, num, false);
                }
            }
        } else {
            return dfs(board, box + 1);
        }
        return false;
    }
    void initialize(const vector<vector<char>>& board) {
        for (int row = 0; row < 9; row++) {
            for (int col = 0; col < 9; col++) {
                if (board[row][col] != '.') {
                    int num = board[row][col] - '0';
                    row_used[row][num] = true;
                    col_used[col][num] = true;

                    int grid_index = (row / 3) * 3 + col / 3;
                    grid_used[grid_index][num] = true;
                }
            }
        }
    }
    void solveSudoku(vector<vector<char>>& board) {
        initialize(board);
        dfs(board, 0);
    }
};
