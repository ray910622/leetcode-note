class Solution {
public:
    vector<int> IP;
    vector<string> ans;
    bool isValidIP(const string& s, int start, int end) {
        if (end >= s.size())
            return false;
        if (start != end && s[start] == '0')
            return false;
        int val = stoi(s.substr(start, end - start + 1));
        if (0 <= val && val <= 255)
            return true;
        else
            return false;
    }

    void addIP() {
        string IPstring;
        for (int i = 0; i < 4; i++) {
            if (i != 0)
                IPstring += ".";
            IPstring += to_string(IP[i]);
        }
        ans.push_back(IPstring);
    }

    void backtracking(const string& s, int start) {
        if (IP.size() == 4) {
            if (start >= s.size())
                addIP();
            return;
        }
        for (int i = start; i < start + 3; i++) {
            if (isValidIP(s, start, i)) {
                string substring = s.substr(start, i - start + 1);
                IP.push_back(stoi(substring));
                backtracking(s, i + 1);
                IP.pop_back();
            }
        }
    }
    vector<string> restoreIpAddresses(string s) {
        if (s.size() < 4 || s.size() > 12)
            return {};
        backtracking(s, 0);
        return ans;
    }
};
