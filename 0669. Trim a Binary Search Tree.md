## Initial Solution
```cpp
class Solution {
public:
    TreeNode* trimBST(TreeNode* root, int low, int high) {
        if (root == nullptr)
            return root;
        if (!(low <= root->val && root->val <= high)) {
            // out of range
            TreeNode *leftTree = trimBST(root->left, low, high),
                     *rightTree = trimBST(root->right, low, high);
            if (leftTree == nullptr)
                return rightTree;
            else
                return leftTree;
        } else {
            // in range
            root->left = trimBST(root->left, low, high);
            root->right = trimBST(root->right, low, high);
            return root;
        }
    }
};
```

## Second Solution
 - When `root->val` is out of range, skip recursing into the subtree that is
  guaranteed to be entirely out of range (by the BST property).
```cpp
class Solution {
public:
    TreeNode* trimBST(TreeNode* root, int low, int high) {
        if (root == nullptr)
            return root;
        if (root->val < low) {
            // trim left
            return trimBST(root->right, low, high);
        }
        if (root->val > high) {
            // trim right
            return trimBST(root->left, low, high);
        }
        // in range
        root->left = trimBST(root->left, low, high);
        root->right = trimBST(root->right, low, high);
        return root;
    }
};
```
