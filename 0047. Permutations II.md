## Initial Solution

```cpp
class Solution {
public:
    vector<vector<int>> ans;
    vector<int> path;
    void backtracking(const vector<int>& nums, vector<bool>& used) {
        if (path.size() == nums.size()) {
            ans.push_back(path);
            return;
        }
        vector<bool> dup_nums(21, false);
        for (int i = 0; i < nums.size(); i++) {
            if (dup_nums[nums[i] + 10])
                continue;
            if (used[i])
                continue;
            path.push_back(nums[i]);
            used[i] = true;
            dup_nums[nums[i] + 10] = true;
            backtracking(nums, used);
            used[i] = false;
            path.pop_back();
        }
    }
    vector<vector<int>> permuteUnique(vector<int>& nums) {
        vector<bool> used(nums.size(), false);
        backtracking(nums, used);
        return ans;
    }
};
```

## Second Solution
- Erase `vector<bool> dup_nums(21, false)`, use `used[]` to check duplicated permutation.
```cpp
class Solution {
public:
    vector<vector<int>> ans;
    vector<int> path;
    void backtracking(const vector<int>& nums, vector<bool>& used) {
        if (path.size() == nums.size()) {
            ans.push_back(path);
            return;
        }
        for (int i = 0; i < nums.size(); i++) {
            if (i > 0 && nums[i] == nums[i - 1] && used[i - 1] == false)
                continue;
            if (used[i])
                continue;
            path.push_back(nums[i]);
            used[i] = true;
            backtracking(nums, used);
            used[i] = false;
            path.pop_back();
        }
    }
    vector<vector<int>> permuteUnique(vector<int>& nums) {
        vector<bool> used(nums.size(), false);
        sort(nums.begin(), nums.end());
        backtracking(nums, used);
        return ans;
    }
};
```
