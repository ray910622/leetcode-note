class Solution {
public:
    int findInorderIndex(vector<int>& vec, int low, int high, int target) {
        for (int i = low; i <= high; i++) {
            if (vec[i] == target) {
                return i;
            }
        }
        return -1;
    }
    TreeNode* buildTreeIdx(vector<int>& inorder, vector<int>& postorder,
                           int inLeft, int inRight, int postLeft,
                           int postRight) {
        // closed interval
        if (postLeft > postRight) {
            return nullptr;
        }

        int rootVal = postorder[postRight];
        int rootIdx = findInorderIndex(inorder, inLeft, inRight, rootVal);
        int leftTreeSize = rootIdx - inLeft, rightTreeSize = inRight - rootIdx;
        TreeNode* root = new TreeNode(rootVal);

        root->left = buildTreeIdx(inorder, postorder, inLeft, rootIdx - 1,
                                  postLeft, postLeft + leftTreeSize - 1);
        root->right = buildTreeIdx(inorder, postorder, rootIdx + 1, inRight,
                                   postRight - rightTreeSize, postRight - 1);

        return root;
    }

    TreeNode* buildTree(vector<int>& inorder, vector<int>& postorder) {

        return buildTreeIdx(inorder, postorder, 0, inorder.size() - 1, 0,
                            postorder.size() - 1);
    }
};
